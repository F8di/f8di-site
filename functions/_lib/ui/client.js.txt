"use strict";
(() => {
  const CSRF = document.querySelector('meta[name="csrf-token"]').content;
  const $ = (s, r = document) => r.querySelector(s);
  const $$ = (s, r = document) => [...r.querySelectorAll(s)];
  const view = $("#view");
  const COLORS = ["#22d3ee", "#3b82f6", "#a78bfa", "#34d399", "#fbbf24", "#f87171", "#f472b6", "#94a3b8"];

  // ---------- helpers ----------
  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[c]);
  const isNum = (x) => typeof x === "number" && Number.isFinite(x);
  const num = (x, d = 0) => (isNum(x) ? x.toLocaleString("en-US", { maximumFractionDigits: d }) : "–");
  const compact = (x) => (isNum(x) ? Intl.NumberFormat("en-US", { notation: "compact", maximumFractionDigits: 1 }).format(x) : "–");
  const pct = (x, d = 1) => (isNum(x) ? `${x >= 0 ? "+" : ""}${(x * 100).toFixed(d)}%` : "–");
  const pctPlain = (x, d = 1) => (isNum(x) ? `${(x * 100).toFixed(d)}%` : "–");
  const cls = (x) => (!isNum(x) ? "flat" : x > 0.005 ? "up" : x < -0.005 ? "down" : "flat");
  const delta = (x) => `<span class="${cls(x)}">${pct(x)}</span>`;
  const dt = (ms) => (ms ? new Date(ms).toLocaleString(undefined, { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit" }) : "–");
  const day = (s) => (s ? new Date(s).toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" }) : "–");
  const ago = (ms) => {
    if (!ms) return "never";
    const s = (Date.now() - ms) / 1000;
    if (s < 60) return "just now";
    if (s < 3600) return `${Math.round(s / 60)}m ago`;
    if (s < 86400) return `${Math.round(s / 3600)}h ago`;
    return `${Math.round(s / 86400)}d ago`;
  };
  const conf = (c) => `<span class="conf" title="Confidence ${isNum(c) ? c.toFixed(2) : "–"}"><i data-w="${Math.round((c || 0) * 100)}"></i></span>`;
  const pattern = (p) => `<span class="pill ${esc((p || "").split(" ")[0])}">${esc(p || "–")}</span>`;
  const icon = (url) => (url ? `<img class="icon" src="${esc(url)}" alt="" loading="lazy">` : `<span class="icon"></span>`);
  const gameLink = (r) => `<span class="name">${icon(r.icon)}<a href="#/game/${r.universeId}">${esc(r.name || `Universe ${r.universeId}`)}</a></span>`;

  function applyDynamic(root = document) {
    for (const el of $$("[data-w]", root)) el.style.width = `${Math.max(0, Math.min(100, Number(el.dataset.w)))}%`;
    for (const el of $$("[data-bg]", root)) el.style.background = el.dataset.bg;
  }

  function toast(msg, err = false) {
    const t = $("#toast");
    t.textContent = msg;
    t.className = `toast${err ? " err" : ""}`;
    t.hidden = false;
    clearTimeout(toast.t);
    toast.t = setTimeout(() => (t.hidden = true), 4000);
  }

  async function api(path, { method = "GET", body } = {}) {
    const res = await fetch(`/stats/api/${path}`, {
      method,
      headers: { "X-CSRF-Token": CSRF, ...(body ? { "Content-Type": "application/json" } : {}) },
      body: body ? JSON.stringify(body) : undefined,
      credentials: "same-origin",
    });
    if (res.status === 401) {
      location.href = "/stats/login";
      throw new Error("Signed out");
    }
    const data = await res.json().catch(() => ({}));
    if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);
    return data;
  }

  const html = (s) => {
    const t = document.createElement("template");
    t.innerHTML = s.trim();
    return t.content.firstElementChild;
  };

  // ---------- range ----------
  const state = { range: localStorage.getItem("stats.range") || "7d", from: "", to: "", exp: localStorage.getItem("stats.exp") || "" };
  const rangeQS = () => (state.range === "custom" && state.from && state.to ? `from=${encodeURIComponent(state.from)}&to=${encodeURIComponent(state.to + "T23:59:59Z")}` : `range=${state.range === "custom" ? "7d" : state.range}`);
  function drawRange() {
    const r = $("#range");
    r.innerHTML = ["24h", "7d", "30d", "90d"].map((k) => `<button data-r="${k}" class="${state.range === k ? "on" : ""}">${k}</button>`).join("") + `<input type="date" id="rf" title="From" value="${esc(state.from)}"><input type="date" id="rt" title="To" value="${esc(state.to)}">`;
    $$("button", r).forEach((b) => (b.onclick = () => { state.range = b.dataset.r; localStorage.setItem("stats.range", state.range); drawRange(); render(); }));
    const onDate = () => { state.from = $("#rf").value; state.to = $("#rt").value; if (state.from && state.to) { state.range = "custom"; drawRange(); render(); } };
    $("#rf").onchange = onDate;
    $("#rt").onchange = onDate;
  }

  // ---------- table ----------
  function table(cols, rows, opts = {}) {
    const wrap = html(`<div><div class="row"><input class="in tf" placeholder="Filter…"><span class="spacer"></span><span class="muted small cnt"></span><button class="btn ghost sm csv">Export CSV</button></div><div class="table-wrap section"><table><thead><tr>${cols.map((c) => `<th class="${c.num ? "num" : ""}" data-k="${esc(c.k)}">${esc(c.label)}</th>`).join("")}</tr></thead><tbody></tbody></table></div></div>`);
    if (opts.noTools) wrap.firstElementChild.hidden = true;
    let sortK = opts.sort ?? null;
    let dir = opts.dir ?? -1;
    let filter = "";
    let shown = [];
    const val = (c, r) => (c.sort ? c.sort(r) : r[c.k]);
    const text = (c, r) => (c.text ? c.text(r) : val(c, r));
    const draw = () => {
      shown = rows.filter((r) => !filter || cols.some((c) => String(text(c, r) ?? "").toLowerCase().includes(filter)));
      if (sortK) {
        const c = cols.find((x) => x.k === sortK);
        shown.sort((a, b) => {
          const x = val(c, a);
          const y = val(c, b);
          if (x === null || x === undefined || Number.isNaN(x)) return 1;
          if (y === null || y === undefined || Number.isNaN(y)) return -1;
          return (x > y ? 1 : x < y ? -1 : 0) * dir;
        });
      }
      const limit = opts.limit || 500;
      $("tbody", wrap).innerHTML = shown.length
        ? shown.slice(0, limit).map((r) => `<tr>${cols.map((c) => `<td class="${c.num ? "num" : ""}">${c.fmt ? c.fmt(r[c.k], r) : esc(r[c.k] ?? "–")}</td>`).join("")}</tr>`).join("")
        : `<tr><td colspan="${cols.length}" class="muted">${esc(opts.empty || "No data yet.")}</td></tr>`;
      $(".cnt", wrap).textContent = `${shown.length} row${shown.length === 1 ? "" : "s"}${shown.length > limit ? ` (showing ${limit})` : ""}`;
      $$("th", wrap).forEach((th) => th.classList.toggle("sorted", th.dataset.k === sortK));
      $$("th", wrap).forEach((th) => th.classList.toggle("asc", th.dataset.k === sortK && dir === 1));
      applyDynamic(wrap);
      if (opts.after) opts.after(wrap, shown);
    };
    $$("th", wrap).forEach((th) => (th.onclick = () => { if (sortK === th.dataset.k) dir = -dir; else { sortK = th.dataset.k; dir = -1; } draw(); }));
    $(".tf", wrap).oninput = (e) => { filter = e.target.value.toLowerCase(); draw(); };
    $(".csv", wrap).onclick = () => {
      const q = (v) => { let s = String(v ?? ""); if (/^[=+\-@]/.test(s)) s = `'${s}`; return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s; };
      const csv = [cols.map((c) => q(c.label)).join(","), ...shown.map((r) => cols.map((c) => q(text(c, r))).join(","))].join("\n");
      const a = document.createElement("a");
      a.href = URL.createObjectURL(new Blob([csv], { type: "text/csv" }));
      a.download = `${opts.name || "export"}.csv`;
      a.click();
      setTimeout(() => URL.revokeObjectURL(a.href), 1000);
    };
    draw();
    return wrap;
  }

  // ---------- charts ----------
  const tip = $("#tooltip");
  function lineChart(series, { height = 220, yFmt = compact, xFmt = dt, empty = "Not enough data yet. Charts fill in as snapshots are collected." } = {}) {
    const box = html(`<div class="chart"></div>`);
    const pts = series.flatMap((s) => s.points.filter((p) => isNum(p.v)));
    if (pts.length < 2) {
      box.innerHTML = `<div class="empty">${esc(empty)}</div>`;
      return box;
    }
    const W = 800, H = height, L = 52, R = 10, T = 10, B = 24;
    const xs = pts.map((p) => p.t), ys = pts.map((p) => p.v);
    const x0 = Math.min(...xs), x1 = Math.max(...xs);
    let y0 = Math.min(...ys), y1 = Math.max(...ys);
    if (y0 === y1) { y0 -= 1; y1 += 1; }
    const pad = (y1 - y0) * 0.08;
    y0 = y0 >= 0 ? Math.max(0, y0 - pad) : y0 - pad;
    y1 += pad;
    const X = (t) => L + ((t - x0) / Math.max(x1 - x0, 1)) * (W - L - R);
    const Y = (v) => T + (1 - (v - y0) / (y1 - y0)) * (H - T - B);
    let svg = `<svg viewBox="0 0 ${W} ${H}" preserveAspectRatio="none" height="${H}">`;
    for (let i = 0; i <= 4; i++) {
      const v = y0 + ((y1 - y0) * i) / 4;
      svg += `<line class="grid-line" x1="${L}" x2="${W - R}" y1="${Y(v)}" y2="${Y(v)}"/><text x="${L - 6}" y="${Y(v) + 3}" text-anchor="end">${esc(yFmt(v))}</text>`;
    }
    const span = x1 - x0;
    const ticks = span <= 0 ? [x0] : [0, 1, 2, 3, 4].map((i) => x0 + (span * i) / 4);
    const tickFmt = (t) => (span < 2 * 864e5 ? new Date(t).toLocaleTimeString(undefined, { hour: "2-digit", minute: "2-digit" }) : new Date(t).toLocaleDateString(undefined, { month: "short", day: "numeric" }));
    ticks.forEach((t, i) => {
      svg += `<text x="${X(t)}" y="${H - 6}" text-anchor="${ticks.length === 1 ? "middle" : i === 0 ? "start" : i === ticks.length - 1 ? "end" : "middle"}">${esc(tickFmt(t))}</text>`;
    });
    series.forEach((s, i) => {
      const p = s.points.filter((q) => isNum(q.v)).sort((a, b) => a.t - b.t);
      if (!p.length) return;
      const color = s.color || COLORS[i % COLORS.length];
      const d = p.map((q, j) => `${j ? "L" : "M"}${X(q.t).toFixed(1)},${Y(q.v).toFixed(1)}`).join("");
      if (series.length === 1) svg += `<path class="area" fill="${color}" d="${d}L${X(p[p.length - 1].t)},${Y(y0)}L${X(p[0].t)},${Y(y0)}Z"/>`;
      svg += `<path class="line" stroke="${color}" d="${d}"/>`;
    });
    svg += `<line class="hover-line" x1="0" x2="0" y1="${T}" y2="${H - B}" visibility="hidden"/></svg>`;
    box.innerHTML = svg + (series.length > 1 ? `<div class="legend">${series.map((s, i) => `<span><i data-bg="${s.color || COLORS[i % COLORS.length]}"></i>${esc(s.name)}</span>`).join("")}</div>` : "");
    applyDynamic(box);
    const el = $("svg", box), hl = $(".hover-line", box);
    el.addEventListener("mousemove", (e) => {
      const rect = el.getBoundingClientRect();
      const t = x0 + (((e.clientX - rect.left) / rect.width) * W - L) / (W - L - R) * (x1 - x0);
      const rows = series.map((s, i) => {
        let best = null;
        for (const q of s.points) if (isNum(q.v) && (!best || Math.abs(q.t - t) < Math.abs(best.t - t))) best = q;
        return best ? { s, i, q: best } : null;
      }).filter(Boolean);
      if (!rows.length) return;
      hl.setAttribute("x1", X(rows[0].q.t)); hl.setAttribute("x2", X(rows[0].q.t)); hl.setAttribute("visibility", "visible");
      tip.innerHTML = `<div class="muted small">${esc(xFmt(rows[0].q.t))}</div>` + rows.map((r) => `<div><span class="${r.i ? "" : ""}">${esc(r.s.name || "value")}</span>: <b>${esc(yFmt(r.q.v))}</b></div>`).join("");
      tip.hidden = false;
      tip.style.left = `${Math.min(e.clientX + 14, innerWidth - 290)}px`;
      tip.style.top = `${e.clientY + 14}px`;
    });
    el.addEventListener("mouseleave", () => { tip.hidden = true; hl.setAttribute("visibility", "hidden"); });
    return box;
  }

  function bars(items, fmt = compact) {
    if (!items.length) return html(`<div class="empty">No data yet.</div>`);
    const max = Math.max(...items.map((i) => Math.abs(i.value) || 0), 1);
    const box = html(`<div class="bars">${items.map((i) => `<div class="bar"><span class="lbl" title="${esc(i.label)}">${esc(i.label)}</span><span class="track"><span class="fill" data-w="${(Math.abs(i.value) / max) * 100}"></span></span><span class="val">${esc(fmt(i.value))}</span></div>`).join("")}</div>`);
    applyDynamic(box);
    return box;
  }

  const card = (title, content, extra = "") => {
    const c = html(`<div class="card"><h2><span>${esc(title)}</span>${extra}</h2></div>`);
    if (typeof content === "string") c.insertAdjacentHTML("beforeend", content);
    else if (content) c.append(content);
    return c;
  };
  const kpi = (label, value, d) => `<div class="card kpi"><div class="l">${esc(label)}</div><div class="v">${value}</div>${d !== undefined ? `<div class="d">${d}</div>` : ""}</div>`;
  const grid = (cls, children) => {
    const g = html(`<div class="grid ${cls} section"></div>`);
    for (const c of children) typeof c === "string" ? g.insertAdjacentHTML("beforeend", c) : g.append(c);
    return g;
  };

  function luau(key, token) {
    return `-- ServerScriptService > F8diStats (ModuleScript). Requires Game Settings > Security > Allow HTTP Requests.
-- Server-only: never put this in ReplicatedStorage. Prefer storing the token as a Roblox Secret.
local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")

local ENDPOINT = "${location.origin}/stats/ingest"
local KEY = "${key}"
local TOKEN = "${token || "<paste token>"}"

local Stats, queue, sessions = {}, {}, {}

local function push(event)
  event.ts = os.time()
  table.insert(queue, event)
end

function Stats.flush()
  if #queue == 0 then return end
  local batch = queue
  queue = {}
  local ok, err = pcall(function()
    HttpService:RequestAsync({
      Url = ENDPOINT, Method = "POST",
      Headers = { ["Content-Type"] = "application/json", ["X-Experience-Key"] = KEY, ["Authorization"] = "Bearer " .. TOKEN },
      Body = HttpService:JSONEncode({ events = batch }),
    })
  end)
  if not ok then warn("F8diStats flush failed: " .. tostring(err)) end
end

function Stats.purchase(player, kind, name, robux) push({ type = "purchase", player = player.UserId, source = kind, name = name, amount = robux }) end -- kind: "product" | "gamepass"
function Stats.currency(player, currency, amount, source) push({ type = "currency", player = player.UserId, currency = currency, amount = amount, source = source }) end -- negative amount = spent
function Stats.item(player, name, action, rarity, qty) push({ type = "item", player = player.UserId, name = name, source = action, rarity = rarity, amount = qty or 1 }) end -- action: "create" | "consume"
function Stats.progress(player, value) push({ type = "progress", player = player.UserId, amount = value }) end
function Stats.custom(player, name, value) push({ type = "custom", player = player and player.UserId, name = name, amount = value }) end

Players.PlayerAdded:Connect(function(p)
  sessions[p] = { id = HttpService:GenerateGUID(false), start = os.clock() }
  push({ type = "session_start", player = p.UserId, session = sessions[p].id })
end)
Players.PlayerRemoving:Connect(function(p)
  local s = sessions[p]
  if s then push({ type = "session_end", player = p.UserId, session = s.id, amount = os.clock() - s.start }) end
  sessions[p] = nil
end)
game:GetService("ScriptContext").Error:Connect(function(message)
  push({ type = "error", name = string.sub(message, 1, 120) })
end)
task.spawn(function()
  while true do
    task.wait(30)
    push({ type = "perf", name = "server_fps", amount = workspace:GetRealPhysicsFPS() })
    Stats.flush()
  end
end)
game:BindToClose(Stats.flush)

return Stats`;
  }

    const cols = res.rows.length ? Object.keys(res.rows[0]).map((k) => ({ k, label: k, num: typeof res.rows[0][k] === "number" && !/ts|hour|_at$/.test(k), fmt: /(^ts$|^hour$|_at$|_ts$)/.test(k) ? (v) => (isNum(v) ? `<span title="${v}">${esc(dt(v))}</span>` : "–") : undefined })) : [{ k: "x", label: "" }];
    root.append(grid("g1", [card(`${ds} · ${res.rows.length} rows`, table(cols, res.rows, { name: ds, empty: "No rows in this range." }))]));

  };

  VIEWS.settings = async () => {
    const [st, se] = await Promise.all([api("status"), api("sessions")]);
    const cfg = st.config, q = st.queue;
    const yes = (b, extra = "") => (b ? `<span class="ok">Configured</span>` : `<span class="muted">Not set</span>${extra}`);
    const root = html(`<div></div>`);
    root.append(grid("g2", [
      card("Configuration", `<dl class="kv"><dt>Database (STATS_DB)</dt><dd>${yes(cfg.database)}</dd><dt>Scheduled collection (STATS_CRON_TOKEN)</dt><dd>${yes(cfg.cronToken, " · collection runs only while the dashboard is open")}</dd><dt>Alert webhook</dt><dd>${yes(cfg.alertWebhook)}</dd><dt>Workers AI (AI)</dt><dd>${yes(cfg.workersAi)}</dd><dt>Open Cloud key</dt><dd>${yes(cfg.openCloudKey)}</dd></dl>

      card("Collection", `${collectionStatus(q)}<div class="row section"><button class="btn" id="collect">Collect now</button><button class="btn ghost" id="retry" ${q.failed.length ? "" : "disabled"}>Retry failed jobs (${q.failed.length})</button></div>`),
    ]));
    $("#collect", root).onclick = async (e) => { e.target.disabled = true; try { const r = await api("collect", { method: "POST" }); toast(`Ran ${r.ran.length} job(s), ${r.ran.filter((x) => !x.ok).length} failed, ${r.remaining} queued.`); render(); } catch (err) { toast(err.message, true); e.target.disabled = false; } };

    root.append(grid("g1", [card("Recent job runs", table([{ k: "job_type", label: "Job" }, { k: "started_at", label: "Started", fmt: dt }, { k: "finished_at", label: "Took", num: true, sort: (r) => r.finished_at - r.started_at, fmt: (v, r) => (v ? `${((v - r.started_at) / 1000).toFixed(1)}s` : "running") }, { k: "ok", label: "Result", fmt: (v) => (v ? `<span class="ok">ok</span>` : v === 0 ? `<span class="bad">failed</span>` : "–") }, { k: "items", label: "Items", num: true }, { k: "error", label: "Error", fmt: (v) => `<span class="small">${esc(v || "")}</span>` }], q.runs, { name: "job-runs" }))]));

  // ---------- routing ----------
  const NAV = [
    ["Analytics", [["overview", "Overview"], ["trends", "Game Trends"], ["explorer", "Game Explorer"], ["competitors", "Competitors"], ["genres", "Genre Analytics"], ["opportunities", "Opportunities"], ["collectibles", "Collectibles"]]],
    ["My games", [["experiences", "My Experiences"], ["economy", "Economy"], ["monetization", "Monetization"], ["retention", "Retention"]]],
    ["Research", [["reports", "AI Reports"], ["alerts", "Alerts"], ["watchlists", "Watchlists"], ["data", "Data Explorer"]]],
    ["", [["settings", "Settings"]]],
  ];
  const TITLES = Object.fromEntries(NAV.flatMap(([, items]) => items));
  Object.assign(TITLES, { game: "Game", item: "Collectible", report: "Report" });
  const PARENT = { game: "explorer", item: "collectibles", report: "reports" };

  function drawNav(current) {
    $("#nav").innerHTML = NAV.map(([g, items]) => `${g ? `<div class="group">${esc(g)}</div>` : `<div class="group"></div>`}${items.map(([k, v]) => `<a href="#/${k}" class="${k === (PARENT[current] || current) ? "active" : ""}"><span>${esc(v)}</span>${k === "alerts" && navAlerts ? `<span class="count">${navAlerts}</span>` : ""}</a>`).join("")}`).join("");
  }
  let navAlerts = 0;
  let renderSeq = 0;

  async function render() {
    const [rawRoute] = location.hash.replace(/^#\/?/, "").split("?");
    const [name = "overview", arg] = rawRoute.split("/");
    const key = VIEWS[name] ? name : "overview";
    drawNav(key);
    $("#page-title").textContent = TITLES[key] || "Overview";
    $("#sidebar").classList.remove("open");
    const seq = ++renderSeq;
    view.innerHTML = `<div class="loading">Loading…</div>`;
    try {
      const node = await VIEWS[key](arg ? decodeURIComponent(arg) : undefined);
      if (seq !== renderSeq) return;
      view.innerHTML = "";
      view.append(node);
      applyDynamic(view);
      view.style.animation = "none";
      void view.offsetWidth;
      view.style.animation = "";
      $("#updated").textContent = `Updated ${new Date().toLocaleTimeString()}`;
    } catch (e) {
      if (seq !== renderSeq) return;
      view.innerHTML = `<div class="empty down">${esc(e.message)}</div>`;
    }
  }

  async function refreshBanner() {
    try {
      const st = await api("status");
      const q = st.queue;
      const errs = q.errors24h.reduce((s, e) => s + e.n, 0);
      const b = $("#banner");
      const stale = !q.lastTickAt || Date.now() - q.lastTickAt > 2 * 3600e3;
      if (errs || stale) {
        b.hidden = false;
        b.className = `banner${errs ? " err" : ""}`;
        b.textContent = [stale ? `Last collection ${ago(q.lastTickAt)}.${st.config.cronToken ? "" : " Scheduled collection isn't configured, so data is only collected while the dashboard is open."}` : "", errs ? `${errs} collection error(s) in the last 24h: ${q.errors24h.map((e) => `${e.job_type} (${e.sample})`).join("; ").slice(0, 300)}` : ""].filter(Boolean).join(" ");
      } else b.hidden = true;
      const a = await api("alerts");
      navAlerts = a.alerts.filter((x) => !x.acknowledged).length;
      drawNav((location.hash.replace(/^#\/?/, "").split(/[/?]/)[0]) || "overview");
    } catch { /* shown on next render */ }
  }

  $("#logout").onclick = async () => { try { await api("logout", { method: "POST" }); } finally { location.href = "/stats/login"; } };
  $("#menu").onclick = () => $("#sidebar").classList.toggle("open");
  addEventListener("hashchange", render);
  drawRange();
  render();
  refreshBanner();
  setInterval(refreshBanner, 120_000);
})();
